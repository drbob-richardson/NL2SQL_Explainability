\title{\textbf{The Alignment Law for Structure-as-Covariance Retrieval}\\[3pt]
\large A surfacing lemma, an exact one-hop law, and a first-order result for the GMRF kernel (Paper A)}
\author{Robert Richardson}\date{\today}
\begin{document}\maketitle

\noindent This note turns the measured characterization---\emph{structure helps iff the graph aligns with the
reasoning chain}---into a proved statement. We give (i) an exact surfacing condition, (ii) an exact one-hop
\emph{alignment law} for the excess surfacing probability, which is zero at zero alignment, positive iff the graph
is chain-assortative, and linear in the alignment $p-q$, and (iii) a first-order alignment result for the
\emph{actual} GMRF kernel used by the method. The oracle-graph ceiling and the adversarial (MuSiQue) boundary are
the two ends of this axis.

\section{Model}
Candidates $[N]$ carry prior means $m_i\in[0,1]$ (calibrated retriever scores) and a positive-definite
correlation-form kernel $K$ ($K_{ii}=1$). Latent relevance is $f\sim\N(m,K)$; judging candidate $j$ reveals $y_j$
under Gaussian noise $\sigma^2$. The active retriever selects judgments by UCB, conditions the GP, and ranks by
posterior mean; recall@$k$ is $|\text{top-}k\cap\text{gold}|/k$.
\begin{assumption}[Buried bridge]\label{ass:bridge}
Gold $=\{a,b\}$ with $m_a=\max_i m_i$ (a \emph{findable anchor}) and $m_b<\min_{d\in D}m_d$ (a \emph{bridge}
buried below the distractors $D$, $|D|=N-2$). Judging is informative, $y_a>m_a$.
\end{assumption}

\section{The surfacing lemma}
\begin{lemma}[Surfacing]\label{lem:surface}
Under Assumption~\ref{ass:bridge}, with unit prior variances the first UCB pick is the anchor $a$; conditioning on
$y_a$ gives $\mu_i=m_i+\tfrac{K_{ia}}{1+\sigma^2}(y_a-m_a)$, and the bridge enters the top-$k$ iff the
\emph{kernel differential} exceeds the burial threshold,
\[
   K_{ba}-\max_{d\in D}K_{da}\ >\ \tau:=\frac{(1+\sigma^2)(\max_d m_d-m_b)}{y_a-m_a}\ >\ 0.
\]
\end{lemma}
\begin{proof}
With $K_{ii}=1$ and equal prior variances $\mathrm{UCB}_i=m_i+\beta$ is maximized at $a$, judged first. The
single-point update is $\mu_i=m_i+K_{ia}(1+\sigma^2)^{-1}(y_a-m_a)$; as $a$ is gold it holds top rank, and $b$
takes the second slot iff $\mu_b>\mu_d\ \forall d$, i.e.\ $(K_{ba}-K_{da})\tfrac{y_a-m_a}{1+\sigma^2}>m_d-m_b$,
which rearranges to the stated threshold at the maximizing $d$.
\end{proof}

\section{The alignment law}
Let the graph be drawn from a stochastic block model: an edge within the chain (the gold pair $\{a,b\}$) with
probability $p$ and any other edge with probability $q$; \emph{alignment} $=p-q$.
\begin{theorem}[Exact one-hop alignment law]\label{thm:onehop}
Consider the one-hop kernel $K=\corr(I+\beta A)$, so $K_{ij}=\beta A_{ij}$ for $i\neq j$, and suppose
$0<\tau<\beta$. Then the bridge surfaces iff $A_{ba}=1$ and $A_{da}=0$ for all $d$, so
\[
   \Pf(\text{surface})=p\,(1-q)^{|D|},
\]
which is strictly increasing in $p$ and strictly decreasing in $q$. Relative to a \emph{density-matched unaligned}
graph (chain edge also present with probability $q$), the \emph{alignment excess} is
\[
   \Delta_{\mathrm{align}}(p,q)\ :=\ \Pf_{p,q}(\text{surface})-\Pf_{q,q}(\text{surface})\ =\ (p-q)\,(1-q)^{|D|},
\]
which is $0$ at $p=q$, strictly positive iff $p>q$, and---holding $q$ fixed---exactly linear in the alignment
$p-q$.
\end{theorem}
\begin{proof}
By Lemma~\ref{lem:surface} and $K_{ij}=\beta A_{ij}$, surfacing requires $\beta(A_{ba}-\max_d A_{da})>\tau$; since
$A\in\{0,1\}$ and $0<\tau<\beta$, this holds iff $A_{ba}=1$ and $\max_d A_{da}=0$. The edges are independent, so
$\Pf(\text{surface})=\Pf(A_{ba}=1)\prod_{d}\Pf(A_{da}=0)=p(1-q)^{|D|}$. The density-matched unaligned graph replaces
$p$ by $q$, giving $q(1-q)^{|D|}$; subtracting yields $(p-q)(1-q)^{|D|}$.
\end{proof}
\begin{remark}[The raw probability does \emph{not} vanish at $p=q$]
At $p=q$ the raw surfacing probability is $q(1-q)^{|D|}>0$: an \emph{unaligned} random graph occasionally surfaces
the bridge by luck. What the alignment law asserts is that the \emph{excess} due to alignment vanishes; this is the
statistically correct statement and explains the small nonzero synthetic gain observed at $p=q$. (An earlier
version claimed $\E[K_{ba}-\max_d K_{da}]=\beta[p-1+(1-q)^{|D|}]$ is zero at $p=q$ and positive iff $p>q$; both are
false because of the maximum-over-distractors penalty. The excess formulation above is the correct object.)
\end{remark}

\begin{proposition}[First-order alignment for the actual GMRF kernel]\label{prop:gmrf}
The method's kernel is the correlation form of the GMRF/CAR covariance $(I+\lambda L)^{-1}$, $L=D-A$. Its
anchor-to-bridge covariance advantage over a random distractor is, to first order,
\[
   \E\big[K_{ba}-K_{da}\big]\ =\ \lambda\,(p-q)\ +\ O(\lambda^2).
\]
\end{proposition}
\begin{proof}
$(I+\lambda L)^{-1}=I-\lambda L+O(\lambda^2)$, and for $i\neq j$ the off-diagonal of $-\lambda L=-\lambda D+\lambda A$
is $\lambda A_{ij}$, so $K_{ij}=\lambda A_{ij}+O(\lambda^2)$; correlation normalization divides by
$\sqrt{K_{ii}K_{jj}}=1+O(\lambda)$ and preserves the first-order off-diagonal term. Taking expectations,
$\E[K_{ba}]=\lambda p+O(\lambda^2)$ and $\E[K_{da}]=\lambda q+O(\lambda^2)$, giving the claim.
\end{proof}

\paragraph{Numerical verification} (\texttt{paperA\_alignment\_sim.py}, $|D|=10$, $q=0.05$). The one-hop
surfacing rate matches $p(1-q)^{|D|}$ to Monte-Carlo error across $p\in[0.05,1]$; the excess matches
$(p-q)(1-q)^{|D|}$, is $\approx 0$ at $p=q$ (empirical $-0.001$), and rises linearly to $+0.57$ at $p-q=0.95$. For
the actual GMRF kernel ($\lambda=0.15$), $\E[K_{ba}-K_{da}]$ is proportional to $p-q$ with slope $\approx\lambda$
(ratio $\approx 0.8$, the balance the $O(\lambda^2)$ term).

\begin{remark}[Measurable alignment]
In practice $p-q$ is estimated by the empirical graph--chain \emph{assortativity}
$\widehat A=\hat p-\hat q$ (gold--gold edge rate minus gold--distractor edge rate); gold-connectivity alone
estimates $\hat p$ only. Theorem~\ref{thm:onehop} explains why $\widehat A$ predicts the gain, why the oracle-graph
ceiling ($q\to0,p\to1$) and the adversarial failure ($p\approx q$) are the two ends of one axis, and why the
effect saturates.
\end{remark}
\begin{remark}[Scope: this is a surfacing result, not an acquisition theorem]
Lemma~\ref{lem:surface} and Theorem~\ref{thm:onehop} describe what happens \emph{conditional on judging the
anchor}. They do not by themselves prove that UCB is the optimal acquisition rule; the empirical superiority of UCB
over value-of-information and lookahead is reported separately and its full explanation is left open.
\end{remark}
